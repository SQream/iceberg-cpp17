52ICEBERG_EXPORT constexpr std::string_view ToString(SnapshotRefType type) noexcept {
53 switch (type) {
54 case SnapshotRefType::kBranch:
55 return "branch";
56 case SnapshotRefType::kTag:
57 return "tag";
58 default:
59 compat::unreachable();
60 }
61 return "unknown"; // This should never be reached
62}
64ICEBERG_EXPORT inline Result<SnapshotRefType> SnapshotRefTypeFromString(
65 std::string_view str) noexcept {
66 if (str == "branch") return SnapshotRefType::kBranch;
67 if (str == "tag") return SnapshotRefType::kTag;
68 return InvalidArgument("Invalid snapshot reference type: {}", str);
69}
