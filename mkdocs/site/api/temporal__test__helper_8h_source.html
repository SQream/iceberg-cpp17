69class TemporalTestHelper {
70 // Helper function to calculate days since Unix epoch (1970-01-01)
71 static constexpr int32_t DaysSinceEpoch(int32_t year, int32_t month, int32_t day) {
72 // Julian day calculation - simplified for post-1970 dates
73 int32_t days = 0;
74
75 // Add days for complete years since 1970
76 for (int32_t y = 1970; y < year; ++y) {
77 days += 365;
78 if (y % 4 == 0 && (y % 100 != 0 || y % 400 == 0)) {
79 days += 1; // leap year
80 }
81 }
82
83 // Add days for complete months in the current year
84 int32_t days_in_month[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
85 if (year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)) {
86 days_in_month[1] = 29; // leap year
87 }
88
89 for (int32_t m = 1; m < month; ++m) {
90 days += days_in_month[m - 1];
91 }
92
93 // Add remaining days
94 days += day - 1; // day is 1-based
95
96 return days;
97 }

106 static int64_t CreateTime(const TimeParts& parts) {
107 using namespace std::chrono;
108 return duration_cast<microseconds>(hours(parts.hour) + minutes(parts.minute) +
109 seconds(parts.second) +
110 microseconds(parts.microsecond))
111 .count();
112 }

115 static int64_t CreateTimestamp(const TimestampParts& parts) {
116 // Calculate days since epoch
117 int32_t days = DaysSinceEpoch(parts.year, parts.month, parts.day);
118
119 // Convert to microseconds since epoch
120 int64_t days_in_micros = static_cast<int64_t>(days) * 24 * 60 * 60 * 1000000LL;
121 int64_t time_in_micros = static_cast<int64_t>(parts.hour) * 60 * 60 * 1000000LL +
122 static_cast<int64_t>(parts.minute) * 60 * 1000000LL +
123 static_cast<int64_t>(parts.second) * 1000000LL +
124 static_cast<int64_t>(parts.microsecond);
125
126 return days_in_micros + time_in_micros;
127 }

130 static int64_t CreateTimestampTz(const TimestampParts& parts) {
131 // Calculate timestamp without timezone first
132 int64_t timestamp_micros = CreateTimestamp(parts);
133
134 // Adjust for timezone offset (convert from local time to UTC)
135 int64_t tz_adjustment_micros = static_cast<int64_t>(parts.tz_offset_minutes) * 60 * 1000000LL;
136
137 return timestamp_micros - tz_adjustment_micros;
138 }
139
141 static int64_t CreateTimestampNanos(const TimestampNanosParts& parts) {
142 // Calculate days since epoch
143 int32_t days = DaysSinceEpoch(parts.year, parts.month, parts.day);
144
145 // Convert to nanoseconds since epoch
146 int64_t days_in_nanos = static_cast<int64_t>(days) * 24 * 60 * 60 * 1000000000LL;
147 int64_t time_in_nanos = static_cast<int64_t>(parts.hour) * 60 * 60 * 1000000000LL +
148 static_cast<int64_t>(parts.minute) * 60 * 1000000000LL +
149 static_cast<int64_t>(parts.second) * 1000000000LL +
150 static_cast<int64_t>(parts.nanosecond);
151
152 return days_in_nanos + time_in_nanos;
153 }
154
156 static int64_t CreateTimestampTzNanos(const TimestampNanosParts& parts) {
157 // Calculate timestamp without timezone first
158 int64_t timestamp_nanos = CreateTimestampNanos(parts);
159
160 // Adjust for timezone offset (convert from local time to UTC)
161 int64_t tz_adjustment_nanos = static_cast<int64_t>(parts.tz_offset_minutes) * 60 * 1000000000LL;
162
163 return timestamp_nanos - tz_adjustment_nanos;
164 }
