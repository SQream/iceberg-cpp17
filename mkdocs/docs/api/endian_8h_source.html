31// C++17 compatible bit_cast replacement
32template <class To, class From>
33typename std::enable_if_t<
34 sizeof(To) == sizeof(From) &&
35 std::is_trivially_copyable_v<From> &&
36 std::is_trivially_copyable_v<To>,
37 To>
38bit_cast(const From& src) noexcept {
39 static_assert(std::is_trivially_constructible_v<To>,
40 "This implementation requires To to be trivially constructible");
41 To dst;
42 std::memcpy(&dst, &src, sizeof(To));
43 return dst;
44}

48// Compile-time endianness detection using a constexpr technique that works in C++17
49constexpr bool is_little_endian_constexpr() {
50 return (__BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__);
51}

53// Manual byteswap implementations
54template <typename T>
55constexpr typename std::enable_if_t<sizeof(T) == 1, T> byteswap_impl(T value) {
56 return value;
57}
58
59template <typename T>
60constexpr typename std::enable_if_t<sizeof(T) == 2, T> byteswap_impl(T value) {
61 uint16_t v = static_cast<uint16_t>(value);
62 v = ((v & 0x00FFU) << 8) | ((v & 0xFF00U) >> 8);
63 return static_cast<T>(v);
64}
65
66template <typename T>
67constexpr typename std::enable_if_t<sizeof(T) == 4, T> byteswap_impl(T value) {
68 uint32_t v = static_cast<uint32_t>(value);
69 v = ((v & 0x000000FFU) << 24) | ((v & 0x0000FF00U) << 8) |
70 ((v & 0x00FF0000U) >> 8) | ((v & 0xFF000000U) >> 24);
71 return static_cast<T>(v);
72}
73
74template <typename T>
75constexpr typename std::enable_if_t<sizeof(T) == 8, T> byteswap_impl(T value) {
76 uint64_t v = static_cast<uint64_t>(value);
77 v = ((v & 0x00000000000000FFULL) << 56) | ((v & 0x000000000000FF00ULL) << 40) |
78 ((v & 0x0000000000FF0000ULL) << 24) | ((v & 0x00000000FF000000ULL) << 8) |
79 ((v & 0x000000FF00000000ULL) >> 8) | ((v & 0x0000FF0000000000ULL) >> 24) |
80 ((v & 0x00FF000000000000ULL) >> 40) | ((v & 0xFF00000000000000ULL) >> 56);
81 return static_cast<T>(v);
82}

87template <typename T>
88constexpr typename std::enable_if_t<std::is_arithmetic_v<T>, T>
89ByteSwap(T value) {
90 if constexpr (sizeof(T) <= 1) {
91 return value;
92 } else if constexpr (std::is_integral_v<T>) {
93 return detail::byteswap_impl(value);
94 } else if constexpr (std::is_floating_point_v<T>) {
95 if constexpr (sizeof(T) == sizeof(uint16_t)) {
96 return bit_cast<T>(detail::byteswap_impl(bit_cast<uint16_t>(value)));
97 } else if constexpr (sizeof(T) == sizeof(uint32_t)) {
98 return bit_cast<T>(detail::byteswap_impl(bit_cast<uint32_t>(value)));
99 } else if constexpr (sizeof(T) == sizeof(uint64_t)) {
100 return bit_cast<T>(detail::byteswap_impl(bit_cast<uint64_t>(value)));
101 } else {
102 static_assert(sizeof(T) == 0,
103 "Unsupported floating-point size for endian conversion.");
104 }
105 }
106}
